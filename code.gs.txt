// ============================================
// TOKEN MANAGEMENT
// ============================================
function generateToken() {
  return Utilities.getUuid();
}

function saveToken(userId, role, schoolId, classId) {
  const token = generateToken();
  const expiry = new Date().getTime() + (TOKEN_EXPIRY_HOURS * 60 * 60 * 1000);
  
  const tokenData = {
    userId: userId,
    role: role,
    schoolId: schoolId,
    classId: classId,
    expiredTime: expiry
  };
  
  const cache = CacheService.getUserCache();
  cache.put(token, JSON.stringify(tokenData), TOKEN_EXPIRY_HOURS * 60 * 60);
  
  return { token: token, expiry: expiry };
}

function validateToken(token) {
  if (!token) {
    return { valid: false, error: 'No token provided' };
  }
  
  const cache = CacheService.getUserCache();
  const tokenDataStr = cache.get(token);
  
  if (!tokenDataStr) {
    return { valid: false, error: 'Invalid or expired token' };
  }
  
  const tokenData = JSON.parse(tokenDataStr);
  
  if (new Date().getTime() > tokenData.expiredTime) {
    cache.remove(token);
    return { valid: false, error: 'Token expired' };
  }
  
  return { valid: true, data: tokenData };
}

function logout(token) {
  if (token) {
    const cache = CacheService.getUserCache();
    cache.remove(token);
  }
  return { success: true, message: 'Logged out successfully' };
}

// AUTHENTICATION
// ============================================
function login(nis, password, schoolId) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const usersSheet = ss.getSheetByName('USERS');
    const data = usersSheet.getDataRange().getValues();
    
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      const userNIS = row[1];
      const userName = row[2];
      const userPass = row[3];
      const userRole = row[4];
      const userSchoolId = row[5];
      const userClassId = row[6];
      const userStatus = row[7];
      const userId = row[0];
      
      // Check credentials
      if (userNIS === nis && userPass === password) {
        // Check status
        if (userStatus !== 'active') {
          return { success: false, error: 'Akun tidak aktif' };
        }
        
        // Super admin can login without school selection
        if (userRole === 'superadmin') {
          const tokenInfo = saveToken(userId, userRole, 'SUPER', '-');
          logActivity(userId, userRole, 'Login', 'SUPER', 'Super admin login');
          
          return {
            success: true,
            token: tokenInfo.token,
            user: {
              userId: userId,
              name: userName,
              role: userRole,
              schoolId: 'SUPER',
              classId: '-'
            }
          };
        }
        
        // Check school match for non-super admin
        if (userSchoolId !== schoolId) {
          return { success: false, error: 'Sekolah tidak sesuai' };
        }
        
        // Check if school is active
        const schoolsSheet = ss.getSheetByName('SCHOOLS');
        const schoolsData = schoolsSheet.getDataRange().getValues();
        let schoolActive = false;
        let schoolName = '';
        
        for (let j = 1; j < schoolsData.length; j++) {
          if (schoolsData[j][0] === schoolId) {
            schoolActive = schoolsData[j][3] === 'active';
            schoolName = schoolsData[j][1];
            break;
          }
        }
        
        if (!schoolActive) {
          return { success: false, error: 'Sekolah tidak aktif' };
        }
        
        const tokenInfo = saveToken(userId, userRole, userSchoolId, userClassId);
        logActivity(userId, userRole, 'Login', userSchoolId, `${userRole} login`);
        
        return {
          success: true,
          token: tokenInfo.token,
          user: {
            userId: userId,
            name: userName,
            role: userRole,
            schoolId: userSchoolId,
            classId: userClassId,
            schoolName: schoolName
          }
        };
      }
    }
    
    return { success: false, error: 'NIS atau password salah' };
    
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

/**
 * Mengambil Data User (Siswa/Guru/Admin)
 * Fitur: Filter Sekolah, Kelas, dan Hitung Total Poin (Ibadah + Kultum)
 */
/**
 * Mengambil Data User (Siswa/Guru/Admin)
 * Fitur: Filter, Hitung Poin, dan MAPPING ID KE NAMA (Sekolah & Kelas)
 */
function getUsers(token, schoolId, classId) {
  const validation = validateToken(token);
  if (!validation.valid) return { success: false, error: validation.error };
  
  if (!['superadmin', 'admin', 'walikelas'].includes(validation.data.role)) {
    return { success: false, error: 'Unauthorized' };
  }
  
  if (['admin', 'walikelas'].includes(validation.data.role) && validation.data.schoolId !== schoolId) {
    return { success: false, error: 'Unauthorized access to school' };
  }

  if (validation.data.role === 'walikelas') {
      classId = validation.data.classId;
  }
  
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // 1. AMBIL SEMUA DATA REFERENSI
    const usersSheet = ss.getSheetByName('USERS');
    const schoolsSheet = ss.getSheetByName('SCHOOLS');
    const classesSheet = ss.getSheetByName('CLASSES');
    const ibadahSheet = ss.getSheetByName('IBADAH');
    const kultumSheet = ss.getSheetByName('KULTUM');

    const usersData = usersSheet.getDataRange().getValues();
    // Handle jika sheet belum ada/kosong
    const schoolsData = schoolsSheet ? schoolsSheet.getDataRange().getValues() : [];
    const classesData = classesSheet ? classesSheet.getDataRange().getValues() : [];
    const ibadahData = ibadahSheet ? ibadahSheet.getDataRange().getValues() : [];
    const kultumData = kultumSheet ? kultumSheet.getDataRange().getValues() : [];

    // ==========================================
    // 2. BUAT MAPPING (KAMUS) ID -> NAMA
    // ==========================================
    
    // Map Sekolah: { 'SCH001': 'SMA 1 JKT', 'SUPER': 'System' }
    let schoolMap = {};
    if (schoolsData.length > 1) {
      for (let i = 1; i < schoolsData.length; i++) {
        schoolMap[schoolsData[i][0]] = schoolsData[i][1]; // ID -> Nama Sekolah
      }
    }

    // Map Kelas: { 'CLS001': 'XII IPA 1' }
    let classMap = {};
    if (classesData.length > 1) {
      for (let i = 1; i < classesData.length; i++) {
        classMap[classesData[i][0]] = classesData[i][2]; // ID -> Nama Kelas
      }
    }

    // ==========================================
    // 3. HITUNG TOTAL POIN
    // ==========================================
    let pointsMap = {};
    
    // Sum Ibadah
    if (ibadahData.length > 1) {
      for (let j = 1; j < ibadahData.length; j++) {
        const uid = String(ibadahData[j][2]);
        const pts = Number(ibadahData[j][12] || 0);
        if (!pointsMap[uid]) pointsMap[uid] = 0;
        pointsMap[uid] += pts;
      }
    }
    // Sum Kultum
    if (kultumData.length > 1) {
      for (let k = 1; k < kultumData.length; k++) {
        const uid = String(kultumData[k][2]);
        const pts = Number(kultumData[k][9] || 0);
        if (!pointsMap[uid]) pointsMap[uid] = 0;
        pointsMap[uid] += pts;
      }
    }
    
    // ==========================================
    // 4. FILTER & SUSUN DATA FINAL
    // ==========================================
    const users = [];
    
    for (let i = 1; i < usersData.length; i++) {
      const currentUserId = String(usersData[i][0]);
      const userSchoolId = usersData[i][5];
      const userClassId = usersData[i][6];
      const userRole = usersData[i][4];
      
      let include = false;

      // Filter Logic (Sama seperti sebelumnya)
      if (validation.data.role === 'superadmin') {
        let filterSchoolId = schoolId;
        if (filterSchoolId === 'SUPER') filterSchoolId = null;

        if (filterSchoolId && classId) include = (userSchoolId === filterSchoolId && userClassId === classId);
        else if (filterSchoolId) include = (userSchoolId === filterSchoolId);
        else include = true;
      } 
      else if (validation.data.role === 'admin') {
        if (classId) include = (userSchoolId === schoolId && userClassId === classId);
        else include = (userSchoolId === schoolId);
      }
      else if (validation.data.role === 'walikelas') {
        include = (userSchoolId === schoolId && userClassId === classId);
      }
      
      if (include) {
        // TERJEMAHKAN ID KE NAMA
        // Jika ID Sekolah 'SUPER', beri nama khusus
        let realSchoolName = (userSchoolId === 'SUPER') ? 'Super Admin System' : (schoolMap[userSchoolId] || userSchoolId);
        let realClassName = classMap[userClassId] || '-';

        users.push({
          userId: currentUserId,
          nis: usersData[i][1],
          name: usersData[i][2],
          role: userRole,
          schoolId: userSchoolId,
          classId: userClassId,
          
          // Data Baru untuk Tampilan:
          schoolName: realSchoolName, 
          className: realClassName,
          
          status: usersData[i][7],
          created: formatDateForDisplay(usersData[i][8]),
          totalPoin: pointsMap[currentUserId] || 0
        });
      }
    }
    
    return { success: true, data: users };
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

// [GANTI FUNGSI addUser LAMA DENGAN INI]
function addUser(token, nis, name, password, role, schoolId, classId) {
  // 1. Validasi Token
  const validation = validateToken(token);
  if (!validation.valid) {
    return { success: false, error: validation.error };
  }
  
  const currentUser = validation.data;

  // 2. Cek Role yang Diizinkan (Superadmin, Admin, Walikelas)
  if (!['superadmin', 'admin', 'walikelas'].includes(currentUser.role)) {
    return { success: false, error: 'Unauthorized: Role tidak diizinkan' };
  }
  
  // 3. Validasi Level Sekolah & Role Target
  if (currentUser.role === 'admin') {
    // Admin Sekolah hanya boleh akses sekolah sendiri
    if (currentUser.schoolId !== schoolId) {
      return { success: false, error: 'Unauthorized: Akses sekolah ditolak' };
    }
    // Admin BOLEH membuat Siswa ATAU Wali Kelas, TAPI TIDAK BOLEH buat Admin lain/Superadmin
    if (role !== 'siswa' && role !== 'walikelas') {
      return { success: false, error: 'Admin hanya bisa membuat akun Siswa atau Wali Kelas' };
    }
  } else if (currentUser.role === 'walikelas') {
     // Wali kelas hanya boleh buat siswa
     if (role !== 'siswa') {
       return { success: false, error: 'Wali kelas hanya bisa membuat akun siswa' };
     }
     // Wali kelas harus berada di sekolah yang sama
     if (currentUser.schoolId !== schoolId) {
        return { success: false, error: 'Unauthorized: Sekolah tidak sesuai' };
     }
  }

  // 4. Validasi Level Kelas (Khusus Wali Kelas)
  if (currentUser.role === 'walikelas') {
    // Wali kelas hanya bisa menambah siswa di kelas yang dia pegang
    if (classId !== currentUser.classId) {
      return { success: false, error: 'Wali kelas hanya bisa menambah siswa di kelasnya sendiri' };
    }
  }
  
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName('USERS');
    const data = sheet.getDataRange().getValues();

    // 5. Cek Duplikasi NIS/Username di Sekolah yang Sama
    // Kita loop data untuk memastikan tidak ada NIS kembar di sekolah tersebut
    for (let i = 1; i < data.length; i++) {
      // Bersihkan tanda kutip (') jika ada pada data lama agar pembandingannya akurat
      let existingNis = String(data[i][1]).replace(/^'/, ''); 
      let existingSchool = data[i][5];

      // Cek kesamaan (NIS sama DAN Sekolah sama)
      if (existingNis == nis && existingSchool == schoolId) {
        return { success: false, error: 'Username/NIS sudah terdaftar di sekolah ini' };
      }
    }
    
    // 6. Generate ID & Format Data
    const userId = 'USER' + new Date().getTime() + Math.floor(Math.random() * 1000);
    
    // [PENTING] Tambahkan tanda kutip satu (') agar dianggap STRING oleh Google Sheets
    // Ini mencegah hilangnya angka 0 di depan (misal: 001) atau format scientific (E+)
    const nisFormatted = "'" + nis;
    const passwordFormatted = "'" + password;

    // 7. Simpan Data User Baru
    sheet.appendRow([
      userId, 
      nisFormatted,      // Kolom 2: NIS (String)
      name,              // Kolom 3: Nama
      passwordFormatted, // Kolom 4: Password (String)
      role,              // Kolom 5: Role
      schoolId,          // Kolom 6: School ID
      classId || '-',    // Kolom 7: Class ID (Strip jika kosong)
      'active',          // Kolom 8: Status
      new Date()         // Kolom 9: Created At
    ]);

    // 8. Catat Log Aktivitas
    logActivity(
      currentUser.userId, 
      currentUser.role, 
      'Add User', 
      schoolId, 
      `Added user: ${name} (${role}) - NIS: ${nis}`
    );
    
    return { success: true, message: 'User berhasil ditambahkan', userId: userId };

  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

      if (data[i][0] === userId) {

      sheet.getRange(rowIndex, 3).setValue(name);

      sheet.getRange(rowIndex, 8).setValue(status);

function getWaliKelasDashboard(token) {
  const validation = validateToken(token);
  if (!validation.valid) return { success: false, error: validation.error };
  
  // Pastikan hanya Wali Kelas yang bisa akses
  if (validation.data.role !== 'walikelas') return { success: false, error: 'Unauthorized' };
  
  try {
    const myClassId = validation.data.classId;
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // 1. Load Semua Sheet yang Diperlukan
    const usersSheet = ss.getSheetByName('USERS');
    const ibadahSheet = ss.getSheetByName('IBADAH');
    const classesSheet = ss.getSheetByName('CLASSES');
    const kultumSheet = ss.getSheetByName('KULTUM'); // Tambahan Sheet Kultum
    
    // 2. Ambil Nama Kelas
    let className = myClassId;
    const classData = classesSheet ? classesSheet.getDataRange().getValues() : [];
    for(let i=1; i<classData.length; i++) {
        if(classData[i][0] === myClassId) {
            className = classData[i][2];
            break;
        }
    }

    // 3. Hitung Jumlah Siswa Aktif di Kelas Ini
    const usersData = usersSheet ? usersSheet.getDataRange().getValues() : [];
    let totalStudents = 0;
    for (let i = 1; i < usersData.length; i++) {
      // Filter: ClassID cocok AND Role='siswa' AND Status='active'
      if (usersData[i][6] === myClassId && usersData[i][4] === 'siswa' && usersData[i][7] === 'active') {
        totalStudents++;
      }
    }
    
    // 4. Hitung Total Poin Kelas (Ibadah + Kultum)
    let totalPoin = 0;
    
    // --- A. Hitung Poin Ibadah ---
    const ibadahData = ibadahSheet ? ibadahSheet.getDataRange().getValues() : [];
    if (ibadahData.length > 1) {
      for (let i = 1; i < ibadahData.length; i++) {
        // Cek ClassID (Kolom E / Index 4)
        if (ibadahData[i][4] === myClassId) {
          totalPoin += Number(ibadahData[i][12] || 0); // Tambah Poin (Kolom M / Index 12)
        }
      }
    }

    // --- B. Hitung Poin Kultum (BARU) ---
    const kultumData = kultumSheet ? kultumSheet.getDataRange().getValues() : [];
    if (kultumData.length > 1) {
      for (let i = 1; i < kultumData.length; i++) {
        // Cek ClassID (Kolom E / Index 4)
        if (kultumData[i][4] === myClassId) {
          totalPoin += Number(kultumData[i][9] || 0); // Tambah Poin (Kolom J / Index 9)
        }
      }
    }
    
    // 5. Ambil Ranking Kelas (Top 5)
    // Menggunakan fungsi getRankingClass yang sudah kita update sebelumnya (sudah include kultum)
    const rankingRes = getRankingClass(token, myClassId);
    const top5 = rankingRes.success ? rankingRes.data.slice(0, 5) : [];
    
    return {
      success: true,
      data: {
        className: className,
        totalStudents: totalStudents,
        totalPoin: totalPoin,
        top5: top5
      }
    };
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

function saveKultumData(token, tanggal, penceramah, judul, masjid, ringkasan) {

  // ============================================================
  const settingsSheet = ss.getSheetByName('SETTINGS');
  let ramadhanStart = null;

  if (settingsSheet) {
     const settingsData = settingsSheet.getDataRange().getValues();
     for (let i = 1; i < settingsData.length; i++) {
       if (settingsData[i][0] === 'ramadhan_start_date') {
         ramadhanStart = new Date(settingsData[i][1]);
         ramadhanStart.setHours(0,0,0,0); // Normalisasi jam ke 00:00
         break;
       }
     }
  }

  // Jika Admin sudah set tanggal, lakukan validasi rentang waktu
  if (ramadhanStart) {
      // Hitung Tanggal Akhir (Start + 29 Hari = 30 Hari Total)
      const ramadhanEnd = new Date(ramadhanStart);
      ramadhanEnd.setDate(ramadhanEnd.getDate() + 29); 
      ramadhanEnd.setHours(23, 59, 59, 999); // Sampai detik terakhir hari ke-30

      // Konversi tanggal input user
      const inputDate = new Date(tanggal);
      inputDate.setHours(12, 0, 0, 0); // Set ke siang hari untuk menghindari bias timezone

      // Cek apakah tanggal input berada DI LUAR rentang Ramadhan?
      if (inputDate < ramadhanStart || inputDate > ramadhanEnd) {
          const startStr = Utilities.formatDate(ramadhanStart, Session.getScriptTimeZone(), "dd MMM");
          const endStr = Utilities.formatDate(ramadhanEnd, Session.getScriptTimeZone(), "dd MMM");
          return { 
            success: false, 
            error: `Jurnal Kultum hanya bisa diisi saat Ramadhan (${startStr} - ${endStr}).` 
          };
      }
  }
  // ============================================================

  const sheet = ss.getSheetByName('KULTUM');
  const data = sheet.getDataRange().getValues();
  const userId = validation.data.userId;

  let rowFound = -1;
  const poinKultum = 10; // Poin fix untuk kultum

  // Cek apakah user sudah isi kultum di tanggal ini? (Untuk Update)
  for (let i = 1; i < data.length; i++) {
    let rowDate = data[i][1];
    // Pastikan format tanggal string yyyy-MM-dd untuk perbandingan
    if (rowDate instanceof Date) rowDate = Utilities.formatDate(rowDate, Session.getScriptTimeZone(), "yyyy-MM-dd");
    
    if (data[i][2] === userId && rowDate === tanggal) {
      rowFound = i + 1; // Baris ditemukan (1-based index)
      break;
    }
  }

  if (rowFound > 0) {
    // === UPDATE DATA KULTUM (Jika siswa mengedit isian) ===
    sheet.getRange(rowFound, 6).setValue(penceramah);
    sheet.getRange(rowFound, 7).setValue(judul);
    sheet.getRange(rowFound, 8).setValue(masjid);
    sheet.getRange(rowFound, 9).setValue(ringkasan);
    // Poin tidak perlu diupdate karena nilainya tetap
    
    return { success: true, message: 'Jurnal Kultum diperbarui' };
  } else {
    // === INSERT DATA BARU (Jika belum pernah isi di tgl tersebut) ===
    const kultumId = 'KLM' + new Date().getTime();
    
    sheet.appendRow([
      kultumId,                   // 1. ID Kultum
      tanggal,                    // 2. Tanggal
      userId,                     // 3. User ID
      validation.data.schoolId,   // 4. School ID
      validation.data.classId,    // 5. Class ID
      penceramah,                 // 6. Nama Penceramah
      judul,                      // 7. Judul Ceramah
      masjid,                     // 8. Nama Masjid
      ringkasan,                  // 9. Isi Ringkasan
      poinKultum,                 // 10. Poin (Default: 10)
      getCurrentTime()            // 11. Waktu Input
    ]);
    
    return { success: true, message: 'Jurnal Kultum disimpan (+10 Poin)' };
  }
}

function getIbadahHistory(token, targetUserId) {
  const validation = validateToken(token);
  if (!validation.valid) return { success: false, error: validation.error };
  
  // Default: Ambil data diri sendiri
  let userIdToFetch = validation.data.userId;
  
  // Jika ada targetUserId (Admin/Wali Kelas ingin melihat data siswa)
  if (targetUserId) {
    // Cek otorisasi
    if (['admin', 'superadmin', 'walikelas'].includes(validation.data.role)) {
       userIdToFetch = targetUserId;
    } else if (targetUserId !== validation.data.userId) {
       return { success: false, error: 'Anda tidak memiliki izin melihat data ini.' };
    }
  }
  
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const ibadahSheet = ss.getSheetByName('IBADAH');
    const kultumSheet = ss.getSheetByName('KULTUM');
    
    // Ambil Data Ibadah
    const dataIbadah = ibadahSheet.getDataRange().getValues();
    const historyMap = {};
    
    // Loop Ibadah
    for (let i = 1; i < dataIbadah.length; i++) {
      if (dataIbadah[i][2] === userIdToFetch) { // Cek UserID
        let tgl = dataIbadah[i][1];
        if (tgl instanceof Date) tgl = Utilities.formatDate(tgl, Session.getScriptTimeZone(), "yyyy-MM-dd");
        
        // Parse Dynamic Data
        let checklist = {};
        try { checklist = JSON.parse(dataIbadah[i][18]); } catch (e) {}
        
        historyMap[tgl] = {
          tanggal: tgl,
          totalPoin: Number(dataIbadah[i][12]),
          checklist: checklist,
          catatan: dataIbadah[i][11],
          hasKultum: false
        };
      }
    }
    
    // Loop Kultum (Gabungkan jika ada di tanggal yang sama)
    const dataKultum = kultumSheet.getDataRange().getValues();
    for (let i = 1; i < dataKultum.length; i++) {
      if (dataKultum[i][2] === userIdToFetch) {
        let tgl = dataKultum[i][1];
        if (tgl instanceof Date) tgl = Utilities.formatDate(tgl, Session.getScriptTimeZone(), "yyyy-MM-dd");
        
        const poinKultum = Number(dataKultum[i][9]);
        
        if (historyMap[tgl]) {
          historyMap[tgl].totalPoin += poinKultum;
          historyMap[tgl].penceramah = dataKultum[i][5];
          historyMap[tgl].judul = dataKultum[i][6];
          historyMap[tgl].hasKultum = true;
        } else {
          historyMap[tgl] = {
            tanggal: tgl, totalPoin: poinKultum, checklist: {}, catatan: '-',
            penceramah: dataKultum[i][5], judul: dataKultum[i][6], hasKultum: true
          };
        }
      }
    }
    
    // Convert ke Array & Sort
    const sortedHistory = Object.values(historyMap).sort((a, b) => new Date(b.tanggal) - new Date(a.tanggal));
    
    return { success: true, data: sortedHistory };
    
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

function getDashboardData(token) {

    if (settingsSheet) {
       const settingsData = settingsSheet.getDataRange().getValues();
       for (let i = 1; i < settingsData.length; i++) {
         if (settingsData[i][0] === 'ramadhan_start_date') {
           ramadhanStartDate = new Date(settingsData[i][1]);
           // Reset jam ke 00:00:00 agar perbandingan tanggal akurat
           ramadhanStartDate.setHours(0,0,0,0); 
           break;
         }
       }
    }
    
    // Jika Admin belum set tanggal, pakai default hari ini agar tidak error
    if (!ramadhanStartDate) {
       ramadhanStartDate = new Date();
       ramadhanStartDate.setHours(0,0,0,0);
    }

    const masterList = getMasterIbadahList(token).data || [];
    const dynamicCounts = {};
    masterList.forEach(item => dynamicCounts[item.key] = 0);

    const ibadahData = ibadahSheet.getDataRange().getValues();
    const userData = usersSheet.getDataRange().getValues();
    const kultumData = kultumSheet ? kultumSheet.getDataRange().getValues() : [];

    // Get User Info
    let userName = '';
    let userClassId = '';
    for (let i = 1; i < userData.length; i++) {
      if (userData[i][0] === validation.data.userId) {
        userName = userData[i][2];
        userClassId = userData[i][6];
        break;
      }
    }
    
    // --- LOGIKA HITUNG POIN & COUNTER (DENGAN FILTER TANGGAL) ---
    let totalPoin = 0;
    let ibadahCount = 0; 
    
    for (let i = 1; i < ibadahData.length; i++) {
      // Filter User
      if (ibadahData[i][2] === validation.data.userId) {
        
        // Filter Tanggal: Hanya hitung data yang tanggalnya >= Tanggal Start Ramadhan
        let rowDate = new Date(ibadahData[i][1]);
        rowDate.setHours(0,0,0,0);
        
        // [PENTING] Hanya hitung jika tanggal input >= Start Ramadhan
        if (rowDate >= ramadhanStartDate) {
            
            // 1. Hitung Hari Terinput
            ibadahCount++;
            
            // 2. Tambah Poin
            totalPoin += Number(ibadahData[i][12]); // Kolom 12 = Total Poin Harian
            
            // 3. Hitung Statistik Per Ibadah (Puasa, Tarawih, dll)
            let rowJson = {};
            try {
              const jsonString = ibadahData[i][18]; // Kolom 18 = Dynamic Data JSON
              if (jsonString && jsonString !== '-') rowJson = JSON.parse(jsonString);
            } catch (e) { rowJson = {}; }

            masterList.forEach(item => {
               if (rowJson[item.key] === true) {
                  dynamicCounts[item.key]++;
               }
            });
        }
      }
    }
    
    // Hitung Poin Kultum (Juga difilter tanggal)
    if (kultumData.length > 0) {
      for (let i = 1; i < kultumData.length; i++) {
        if (kultumData[i][2] === validation.data.userId) {
           let kDate = new Date(kultumData[i][1]);
           kDate.setHours(0,0,0,0);
           
           if (kDate >= ramadhanStartDate) {
              totalPoin += Number(kultumData[i][9] || 0);
           }
        }
      }
    }
    
    // Get Rank
    const classRanking = getRankingClass(token, userClassId);
    let rank = 0;
    if (classRanking.success) {
      for (let i = 0; i < classRanking.data.length; i++) {
        if (classRanking.data[i].userId === validation.data.userId) {
          rank = i + 1;
          break;
        }
      }
    }

    // ============================================
    // [FIX] LOGIKA STREAK DENGAN FILTER TANGGAL
    // ============================================
    let currentStreak = 0;
    try {
        // Panggil helper function dengan parameter tanggal mulai
        // Pastikan helper calculateUserStreak() di bagian bawah file juga sudah menerima 3 parameter
        currentStreak = calculateUserStreak(
            validation.data.userId, 
            ibadahSheet.getDataRange().getValues(), 
            ramadhanStartDate // <--- Parameter Kunci: Tanggal batas
        );
    } catch(e) {
        currentStreak = 0; 
    }

    let userBadges = [];
    if (badgeSheet) {
        const badgeData = badgeSheet.getDataRange().getValues();
        for(let i=1; i<badgeData.length; i++) {
           if(badgeData[i][1] === validation.data.userId) {
               userBadges.push(badgeData[i][2]);
           }
        }
    }
    
    return {
      success: true,
      data: {
        userName: userName,
        totalPoin: totalPoin,
        ibadahCount: ibadahCount,
        rank: rank,
        chartData: dynamicCounts,
        masterList: masterList,
        streak: currentStreak,   
        badges: userBadges
      }
    };
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

// HELPER: HITUNG STREAK (ANTI-CURANG / TANGGAL FILTER)
// ============================================
function calculateUserStreak(userId, ibadahData, limitDate) {
  // 1. Tentukan Batas Waktu (00:00:00)
  // Jika limitDate tidak ada, anggap 0 (ambil semua)
  let minTime = 0;
  if (limitDate instanceof Date) {
     limitDate.setHours(0,0,0,0);
     minTime = limitDate.getTime();
  }

  // 2. Ambil & Filter Tanggal User
  const userDates = ibadahData
    .filter(row => row[2] === userId) // Ambil data user yang sesuai (Kolom C)
    .map(row => {
       // Konversi tanggal database ke format waktu (jam 00:00:00)
       const d = new Date(row[1]); 
       d.setHours(0,0,0,0);
       return d.getTime();
    })
    .filter(time => time >= minTime) // [KUNCI ANTI-CURANG] Hapus data sebelum Start Ramadhan
    .sort((a, b) => b - a); // Urutkan dari Terbaru ke Terlama

  // Hapus tanggal duplikat
  const uniqueDates = [...new Set(userDates)]; 

  if (uniqueDates.length === 0) return 0;

  // 3. Setup Hari Ini dan Kemarin
  let today = new Date();
  today.setHours(0,0,0,0);
  const todayTime = today.getTime();
  
  let yesterday = new Date(today);
  yesterday.setDate(yesterday.getDate() - 1);
  const yesterdayTime = yesterday.getTime();

  // 4. Cek Status Streak
  // Streak putus jika user terakhir mengisi sebelum kemarin
  const lastEntryTime = uniqueDates[0];
  
  if (lastEntryTime !== todayTime && lastEntryTime !== yesterdayTime) {
     return 0; 
  }

  // 5. Hitung Mundur (Streak Counter)
  let streak = 1;
  let currentDateCheck = lastEntryTime; 

  for (let i = 1; i < uniqueDates.length; i++) {
    const prevDateLog = uniqueDates[i];
    
    // Harusnya tanggal sebelumnya adalah (currentDateCheck - 1 hari)
    const expectedPrevDate = new Date(currentDateCheck);
    expectedPrevDate.setDate(expectedPrevDate.getDate() - 1); 

    if (prevDateLog === expectedPrevDate.getTime()) {
      streak++;
      currentDateCheck = prevDateLog; // Mundur satu hari
    } else {
      break; // Tanggal tidak urut (bolong), stop menghitung
    }
  }

  return streak;
}

function getSchoolDashboard(token) {
  const validation = validateToken(token);
  if (!validation.valid) {
    return { success: false, error: validation.error };
  }
  
  if (validation.data.role !== 'admin') {
    return { success: false, error: 'Unauthorized' };
  }
  
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // 1. Load Semua Sheet
    const usersSheet = ss.getSheetByName('USERS');
    const classesSheet = ss.getSheetByName('CLASSES');
    const ibadahSheet = ss.getSheetByName('IBADAH');
    const kultumSheet = ss.getSheetByName('KULTUM'); // Tambahan Sheet Kultum
    
    // 2. Ambil Data (Handle jika sheet kosong)
    const usersData = usersSheet ? usersSheet.getDataRange().getValues() : [];
    const classesData = classesSheet ? classesSheet.getDataRange().getValues() : [];
    const ibadahData = ibadahSheet ? ibadahSheet.getDataRange().getValues() : [];
    const kultumData = kultumSheet ? kultumSheet.getDataRange().getValues() : [];
    
    // 3. Hitung Jumlah Siswa Aktif
    let totalStudents = 0;
    if (usersData.length > 1) {
      for (let i = 1; i < usersData.length; i++) {
        // Kolom F (Index 5) = SchoolID, Kolom E (Index 4) = Role, Kolom H (Index 7) = Status
        if (usersData[i][5] === validation.data.schoolId && usersData[i][4] === 'siswa' && usersData[i][7] === 'active') {
          totalStudents++;
        }
      }
    }
    
    // 4. Hitung Jumlah Kelas
    let totalClasses = 0;
    if (classesData.length > 1) {
      for (let i = 1; i < classesData.length; i++) {
        // Kolom B (Index 1) = SchoolID
        if (classesData[i][1] === validation.data.schoolId) {
          totalClasses++;
        }
      }
    }
    
    // 5. Hitung Total Poin (Ibadah + Kultum)
    let totalPoin = 0;
    
    // --- A. Sum dari IBADAH ---
    if (ibadahData.length > 1) {
      for (let i = 1; i < ibadahData.length; i++) {
        // Kolom D (Index 3) = SchoolID, Kolom M (Index 12) = TotalPoin
        if (ibadahData[i][3] === validation.data.schoolId) {
          totalPoin += Number(ibadahData[i][12] || 0);
        }
      }
    }

    // --- B. Sum dari KULTUM (BARU) ---
    if (kultumData.length > 1) {
      for (let i = 1; i < kultumData.length; i++) {
        // Kolom D (Index 3) = SchoolID, Kolom J (Index 9) = Poin
        if (kultumData[i][3] === validation.data.schoolId) {
          totalPoin += Number(kultumData[i][9] || 0);
        }
      }
    }
    
    // 6. Get Top 10 Students (Ranking Sekolah)
    // Menggunakan fungsi ranking yang sudah diupdate sebelumnya
    const ranking = getRankingSchool(token, validation.data.schoolId);
    const top10 = ranking.success ? ranking.data.slice(0, 10) : [];
    
    return {
      success: true,
      data: {
        totalStudents: totalStudents,
        totalClasses: totalClasses,
        totalPoin: totalPoin,
        top10: top10
      }
    };
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

function getGlobalDashboard(token) {
  const validation = validateToken(token);
  if (!validation.valid) {
    return { success: false, error: validation.error };
  }
  
  if (validation.data.role !== 'superadmin') {
    return { success: false, error: 'Unauthorized' };
  }
  
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // 1. Load Semua Sheet
    const schoolsSheet = ss.getSheetByName('SCHOOLS');
    const usersSheet = ss.getSheetByName('USERS');
    const ibadahSheet = ss.getSheetByName('IBADAH');
    const kultumSheet = ss.getSheetByName('KULTUM'); // Tambahan Sheet Kultum
    
    // 2. Ambil Data (Handle jika sheet kosong/null)
    const schoolsData = schoolsSheet ? schoolsSheet.getDataRange().getValues() : [];
    const usersData = usersSheet ? usersSheet.getDataRange().getValues() : [];
    const ibadahData = ibadahSheet ? ibadahSheet.getDataRange().getValues() : [];
    const kultumData = kultumSheet ? kultumSheet.getDataRange().getValues() : [];
    
    // 3. Hitung Jumlah Sekolah (Exclude Super Admin)
    let totalSchools = 0;
    if (schoolsData.length > 1) {
      for (let i = 1; i < schoolsData.length; i++) {
        if (schoolsData[i][0] !== 'SUPER') {
          totalSchools++;
        }
      }
    }
    
    // 4. Hitung Jumlah Siswa Aktif
    let totalStudents = 0;
    if (usersData.length > 1) {
      for (let i = 1; i < usersData.length; i++) {
        // Kolom E (Index 4) = Role, Kolom H (Index 7) = Status
        if (usersData[i][4] === 'siswa' && usersData[i][7] === 'active') {
          totalStudents++;
        }
      }
    }
    
    // 5. Hitung Total Poin & Ranking Sekolah (Ibadah + Kultum)
    let totalPoin = 0;
    const schoolRanking = {}; // Map SchoolID -> Points
    
    // --- A. Sum dari IBADAH ---
    if (ibadahData.length > 1) {
      for (let i = 1; i < ibadahData.length; i++) {
        const p = Number(ibadahData[i][12] || 0); // Kolom M (Index 12)
        totalPoin += p;
        
        const schoolId = ibadahData[i][3]; // Kolom D (Index 3)
        if (!schoolRanking[schoolId]) {
          schoolRanking[schoolId] = 0;
        }
        schoolRanking[schoolId] += p;
      }
    }
    
    // --- B. Sum dari KULTUM (BARU) ---
    if (kultumData.length > 1) {
      for (let i = 1; i < kultumData.length; i++) {
        const p = Number(kultumData[i][9] || 0); // Kolom J (Index 9)
        totalPoin += p;
        
        const schoolId = kultumData[i][3]; // Kolom D (Index 3)
        if (!schoolRanking[schoolId]) {
          schoolRanking[schoolId] = 0;
        }
        schoolRanking[schoolId] += p;
      }
    }
    
    // 6. Format Data Ranking Sekolah (Gabungkan dengan Nama Sekolah)
    const schoolRankingArray = [];
    for (const schoolId in schoolRanking) {
      let schoolName = '';
      // Cari nama sekolah
      for (let i = 1; i < schoolsData.length; i++) {
        if (schoolsData[i][0] === schoolId) {
          schoolName = schoolsData[i][1];
          break;
        }
      }
      
      schoolRankingArray.push({
        schoolId: schoolId,
        schoolName: schoolName,
        totalPoin: schoolRanking[schoolId]
      });
    }
    
    // Urutkan Sekolah berdasarkan Poin Tertinggi
    schoolRankingArray.sort((a, b) => b.totalPoin - a.totalPoin);
    
    // 7. Ambil Top 10 Global Student Ranking
    // Fungsi ini diasumsikan sudah diupdate juga sebelumnya
    const globalRanking = getRankingGlobal(token);
    const top10Global = globalRanking.success ? globalRanking.data.slice(0, 10) : [];
    
    return {
      success: true,
      data: {
        totalSchools: totalSchools,
        totalStudents: totalStudents,
        totalPoin: totalPoin,
        schoolRanking: schoolRankingArray,
        top10Global: top10Global
      }
    };
    
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

function getRankingClass(token, classId) {
  const validation = validateToken(token);
  if (!validation.valid) {
    return { success: false, error: validation.error };
  }
  
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // 1. Ambil Referensi Sheet
    const ibadahSheet = ss.getSheetByName('IBADAH');
    const usersSheet = ss.getSheetByName('USERS');
    const kultumSheet = ss.getSheetByName('KULTUM'); // Tambahan Sheet Kultum
    
    // 2. Ambil Data (Handle jika sheet belum ada)
    const ibadahData = ibadahSheet ? ibadahSheet.getDataRange().getValues() : [];
    const usersData = usersSheet ? usersSheet.getDataRange().getValues() : [];
    const kultumData = kultumSheet ? kultumSheet.getDataRange().getValues() : [];
    
    // Calculate points per user in class
    const userPoints = {};
    
    // --- STEP A: Hitung Poin dari IBADAH ---
    // Mulai loop dari 1 (skip header)
    if (ibadahData.length > 1) {
      for (let i = 1; i < ibadahData.length; i++) {
        // Cek apakah ClassID (Kolom E / Index 4) cocok
        if (ibadahData[i][4] === classId) {
          const userId = ibadahData[i][2]; // UserID (Kolom C)
          
          if (!userPoints[userId]) {
            userPoints[userId] = 0;
          }
          // Tambahkan Poin Ibadah (Kolom M / Index 12)
          userPoints[userId] += Number(ibadahData[i][12] || 0);
        }
      }
    }
    
    // --- STEP B: Hitung Poin dari KULTUM (BARU) ---
    if (kultumData.length > 1) {
      for (let i = 1; i < kultumData.length; i++) {
        // Cek apakah ClassID (Kolom E / Index 4) cocok
        if (kultumData[i][4] === classId) {
          const userId = kultumData[i][2]; // UserID (Kolom C)
          
          if (!userPoints[userId]) {
            userPoints[userId] = 0;
          }
          // Tambahkan Poin Kultum (Kolom J / Index 9)
          userPoints[userId] += Number(kultumData[i][9] || 0);
        }
      }
    }
    
    // --- STEP C: Ambil Detail Siswa (Nama & NIS) ---
    const ranking = [];
    for (const userId in userPoints) {
      let userName = '';
      let nis = '';
      
      // Cari data siswa di sheet USERS
      for (let i = 1; i < usersData.length; i++) {
        if (usersData[i][0] === userId) {
          userName = usersData[i][2]; // Nama
          nis = usersData[i][1];      // NIS
          break;
        }
      }
      
      ranking.push({
        userId: userId,
        nis: nis,
        name: userName,
        totalPoin: userPoints[userId]
      });
    }
    
    // --- STEP D: Urutkan dari Poin Tertinggi ---
    ranking.sort((a, b) => b.totalPoin - a.totalPoin);
    
    return { success: true, data: ranking };
    
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

function getRankingSchool(token, schoolId) {
  const validation = validateToken(token);
  if (!validation.valid) {
    return { success: false, error: validation.error };
  }
  
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // 1. Ambil Referensi Sheet
    const ibadahSheet = ss.getSheetByName('IBADAH');
    const usersSheet = ss.getSheetByName('USERS');
    const classesSheet = ss.getSheetByName('CLASSES');
    const kultumSheet = ss.getSheetByName('KULTUM'); // Tambahan Sheet Kultum
    
    // 2. Ambil Data (Handle jika sheet belum ada/kosong)
    const ibadahData = ibadahSheet ? ibadahSheet.getDataRange().getValues() : [];
    const usersData = usersSheet ? usersSheet.getDataRange().getValues() : [];
    const classesData = classesSheet ? classesSheet.getDataRange().getValues() : [];
    const kultumData = kultumSheet ? kultumSheet.getDataRange().getValues() : [];
    
    // Calculate points per user in school
    const userPoints = {};
    
    // --- STEP A: Hitung Poin dari IBADAH ---
    // Mulai loop dari 1 (skip header)
    if (ibadahData.length > 1) {
      for (let i = 1; i < ibadahData.length; i++) {
        // Cek apakah SchoolID (Kolom D / Index 3) cocok
        if (ibadahData[i][3] === schoolId) {
          const userId = ibadahData[i][2]; // UserID (Kolom C)
          
          if (!userPoints[userId]) {
            userPoints[userId] = 0;
          }
          // Tambahkan Poin Ibadah (Kolom M / Index 12)
          userPoints[userId] += Number(ibadahData[i][12] || 0);
        }
      }
    }
    
    // --- STEP B: Hitung Poin dari KULTUM (BARU) ---
    if (kultumData.length > 1) {
      for (let i = 1; i < kultumData.length; i++) {
        // Cek apakah SchoolID (Kolom D / Index 3) cocok
        if (kultumData[i][3] === schoolId) {
          const userId = kultumData[i][2]; // UserID (Kolom C)
          
          if (!userPoints[userId]) {
            userPoints[userId] = 0;
          }
          // Tambahkan Poin Kultum (Kolom J / Index 9)
          userPoints[userId] += Number(kultumData[i][9] || 0);
        }
      }
    }
    
    // --- STEP C: Ambil Detail Siswa (Nama, NIS, Kelas) ---
    const ranking = [];
    for (const userId in userPoints) {
      let userName = '';
      let nis = '';
      let className = '';
      
      // Cari data siswa di sheet USERS
      for (let i = 1; i < usersData.length; i++) {
        if (usersData[i][0] === userId) {
          userName = usersData[i][2]; // Nama
          nis = usersData[i][1];      // NIS
          
          // Get class name
          const classId = usersData[i][6]; // ClassID
          for (let j = 1; j < classesData.length; j++) {
            if (classesData[j][0] === classId) {
              className = classesData[j][2]; // Nama Kelas
              break;
            }
          }
          break;
        }
      }
      
      ranking.push({
        userId: userId,
        nis: nis,
        name: userName,
        className: className,
        totalPoin: userPoints[userId]
      });
    }
    
    // --- STEP D: Urutkan dari Poin Tertinggi ---
    ranking.sort((a, b) => b.totalPoin - a.totalPoin);
    
    return { success: true, data: ranking };
    
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

function getRankingGlobal(token) {
  const validation = validateToken(token);
  if (!validation.valid) {
    return { success: false, error: validation.error };
  }
  
  // Validasi Role: Hanya Superadmin yang boleh lihat Global Ranking
  if (validation.data.role !== 'superadmin') {
    return { success: false, error: 'Unauthorized' };
  }
  
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // 1. Ambil Referensi Semua Sheet
    const ibadahSheet = ss.getSheetByName('IBADAH');
    const usersSheet = ss.getSheetByName('USERS');
    const schoolsSheet = ss.getSheetByName('SCHOOLS');
    const classesSheet = ss.getSheetByName('CLASSES');
    const kultumSheet = ss.getSheetByName('KULTUM'); // Tambahan Sheet Kultum
    
    // 2. Ambil Data (Handle jika sheet kosong/null)
    const ibadahData = ibadahSheet ? ibadahSheet.getDataRange().getValues() : [];
    const usersData = usersSheet ? usersSheet.getDataRange().getValues() : [];
    const schoolsData = schoolsSheet ? schoolsSheet.getDataRange().getValues() : [];
    const classesData = classesSheet ? classesSheet.getDataRange().getValues() : [];
    const kultumData = kultumSheet ? kultumSheet.getDataRange().getValues() : [];
    
    // 3. Hitung Poin Global
    const userPoints = {};
    
    // --- STEP A: Hitung Poin dari IBADAH ---
    if (ibadahData.length > 1) {
      for (let i = 1; i < ibadahData.length; i++) {
        const userId = ibadahData[i][2]; // Kolom C
        const points = Number(ibadahData[i][12] || 0); // Kolom M (Total Poin)
        
        if (!userPoints[userId]) {
          userPoints[userId] = 0;
        }
        userPoints[userId] += points;
      }
    }
    
    // --- STEP B: Hitung Poin dari KULTUM (BARU) ---
    if (kultumData.length > 1) {
      for (let i = 1; i < kultumData.length; i++) {
        const userId = kultumData[i][2]; // Kolom C
        const points = Number(kultumData[i][9] || 0); // Kolom J (Poin Kultum)
        
        if (!userPoints[userId]) {
          userPoints[userId] = 0;
        }
        userPoints[userId] += points;
      }
    }
    
    // 4. Ambil Detail User (Nama, Sekolah, Kelas)
    const ranking = [];
    for (const userId in userPoints) {
      let userName = '';
      let nis = '';
      let schoolName = '';
      let className = '';
      
      // Cari data user
      for (let i = 1; i < usersData.length; i++) {
        if (usersData[i][0] === userId) {
          userName = usersData[i][2];
          nis = usersData[i][1];
          const schoolId = usersData[i][5];
          const classId = usersData[i][6];
          
          // Cari Nama Sekolah
          for (let j = 1; j < schoolsData.length; j++) {
            if (schoolsData[j][0] === schoolId) {
              schoolName = schoolsData[j][1];
              break;
            }
          }
          
          // Cari Nama Kelas
          for (let k = 1; k < classesData.length; k++) {
            if (classesData[k][0] === classId) {
              className = classesData[k][2];
              break;
            }
          }
          break; // Stop loop user setelah ketemu
        }
      }
      
      ranking.push({
        userId: userId,
        nis: nis,
        name: userName,
        schoolName: schoolName,
        className: className,
        totalPoin: userPoints[userId]
      });
    }
    
    // 5. Urutkan Ranking (Tertinggi ke Terendah)
    ranking.sort((a, b) => b.totalPoin - a.totalPoin);
    
    return { success: true, data: ranking };
    
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

function getActivityLog(token, schoolId, limit) {
  const validation = validateToken(token);
  if (!validation.valid) {
    return { success: false, error: validation.error };
  }
  
  if (!['superadmin', 'admin'].includes(validation.data.role)) {
    return { success: false, error: 'Unauthorized' };
  }
  
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName('LOG_ACTIVITY');
    const data = sheet.getDataRange().getValues();
    
    const logs = [];
    for (let i = data.length - 1; i >= 1; i--) {
      if (validation.data.role === 'superadmin' || data[i][4] === schoolId) {
        logs.push({
          timestamp: formatDateForDisplay(data[i][0]),
          userId: data[i][1],
          role: data[i][2],
          aktivitas: data[i][3],
          schoolId: data[i][4],
          details: data[i][5]
        });
        
        if (limit && logs.length >= limit) break;
      }
    }
    
    return { success: true, data: logs };
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

function exportReport(token, type, schoolId, classId) {
  const validation = validateToken(token);
  if (!validation.valid) {
    return { success: false, error: validation.error };
  }
  
  if (!['superadmin', 'admin'].includes(validation.data.role)) {
    return { success: false, error: 'Unauthorized' };
  }
  
  // Validate school access for admin
  if (validation.data.role === 'admin' && validation.data.schoolId !== schoolId) {
    return { success: false, error: 'Unauthorized access to school' };
  }
  
  try {
    let reportData = {};
    
    if (type === 'class') {
      const ranking = getRankingClass(token, classId);
      reportData = {
        type: 'class',
        data: ranking.data
      };
    } else if (type === 'school') {
      const ranking = getRankingSchool(token, schoolId);
      reportData = {
        type: 'school',
        data: ranking.data
      };
    } else if (type === 'global' && validation.data.role === 'superadmin') {
      const ranking = getRankingGlobal(token);
      reportData = {
        type: 'global',
        data: ranking.data
      };
    }
    
    logActivity(validation.data.userId, validation.data.role, 'Export Report', schoolId || 'SUPER', `Exported ${type} report`);
    
    return { success: true, data: reportData };
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

function saveChecklistData(token, tanggal, checklistDataJSON, catatan) {
  // 1. Validasi Token Login
  const validation = validateToken(token);
  if (!validation.valid) return { success: false, error: validation.error };
  
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // ============================================================
  // [BARU] VALIDASI TANGGAL RAMADHAN (HANYA 30 HARI)
  // ============================================================
  const settingsSheet = ss.getSheetByName('SETTINGS');
  let ramadhanStart = null;
  
  // Ambil tanggal mulai dari Settings
  if (settingsSheet) {
     const settingsData = settingsSheet.getDataRange().getValues();
     for (let i = 1; i < settingsData.length; i++) {
       if (settingsData[i][0] === 'ramadhan_start_date') {
         ramadhanStart = new Date(settingsData[i][1]);
         ramadhanStart.setHours(0,0,0,0); // Set jam ke 00:00
         break;
       }
     }
  }

  // Jika Admin sudah set tanggal, lakukan validasi Range Ramadhan
  if (ramadhanStart) {
      const ramadhanEnd = new Date(ramadhanStart);
      ramadhanEnd.setDate(ramadhanEnd.getDate() + 29); 
      ramadhanEnd.setHours(23, 59, 59, 999); 

      // Konversi tanggal input user
      const inputDateCheck = new Date(tanggal);
      inputDateCheck.setHours(12, 0, 0, 0); 

      if (inputDateCheck < ramadhanStart || inputDateCheck > ramadhanEnd) {
          const startStr = Utilities.formatDate(ramadhanStart, Session.getScriptTimeZone(), "dd MMM");
          const endStr = Utilities.formatDate(ramadhanEnd, Session.getScriptTimeZone(), "dd MMM");
          return { 
            success: false, 
            error: `Gagal: Pengisian hanya diizinkan saat Ramadhan (${startStr} - ${endStr}).` 
          };
      }
  }

  // ============================================================
  // [FITUR BARU] LOCK RETROACTIVE ENTRY (BATAS INPUT MUNDUR)
  // ============================================================
  // Konfigurasi: Berapa hari maksimal siswa boleh mengisi mundur?
  // 0 = Hanya Hari Ini
  // 1 = Hari Ini & Kemarin
  // 2 = Hari Ini, Kemarin, & 2 Hari Lalu
  const MAX_RETROACTIVE_DAYS = 1; 

  // 1. Ambil Tanggal Hari Ini (Server Time)
  const today = new Date();
  // Format ke string YYYY-MM-DD sesuai TimeZone Script agar akurat
  const todayStr = Utilities.formatDate(today, Session.getScriptTimeZone(), "yyyy-MM-dd");
  // Buat object Date yang dinormalisasi ke jam 12:00 siang
  const dateServer = new Date(todayStr);
  dateServer.setHours(12, 0, 0, 0);

  // 2. Ambil Tanggal Input User
  const dateInput = new Date(tanggal);
  dateInput.setHours(12, 0, 0, 0);

  // 3. Hitung Selisih Hari
  // (Server - Input) -> Jika positif berarti masa lalu
  const diffTime = dateServer.getTime() - dateInput.getTime();
  const diffDays = Math.floor(diffTime / (1000 * 60 * 60 * 24));

  // 4. Validasi Batas Mundur
  // Jika selisih hari lebih besar dari batas yang ditentukan
  if (diffDays > MAX_RETROACTIVE_DAYS) {
      return { 
        success: false, 
        error: `Akses Dikunci: Anda tidak dapat mengisi data lebih dari ${MAX_RETROACTIVE_DAYS} hari ke belakang. Batas pengisian terakhir adalah tanggal ${Utilities.formatDate(new Date(dateServer.setDate(dateServer.getDate() - MAX_RETROACTIVE_DAYS)), Session.getScriptTimeZone(), "dd MMM yyyy")}.` 
      };
  }
  
  // Opsional: Blokir Pengisian Masa Depan (Cheating Prevention)
  if (diffDays < 0) {
      return {
        success: false,
        error: `Akses Dikunci: Anda tidak dapat mengisi checklist untuk tanggal masa depan.`
      };
  }
  // ============================================================


  const sheet = ss.getSheetByName('IBADAH');
  const masterSheet = ss.getSheetByName('MASTER_IBADAH');
  
  const userId = validation.data.userId;
  const schoolId = validation.data.schoolId;
  const checklistData = JSON.parse(checklistDataJSON); 
  
  // 2. HITUNG TOTAL POIN (Berdasarkan Master)
  const masterData = masterSheet.getDataRange().getValues();
  let checklistPoin = 0;
  
  for (let i = 1; i < masterData.length; i++) {
    const key = masterData[i][1];
    const poin = Number(masterData[i][2]);
    const status = masterData[i][5];
    
    if (checklistData[key] === true && status === 'active') {
      checklistPoin += poin;
    }
  }

  // 3. SIMPAN KE DATABASE (Update atau Insert)
  const data = sheet.getDataRange().getValues();
  let rowFound = -1;
  let currentKultumPoin = 0; 

  const inputDateStr = String(tanggal);
  for (let i = 1; i < data.length; i++) {
    let rowDate = data[i][1];
    let rowDateStr = '';
    
    if (rowDate instanceof Date) {
      rowDateStr = Utilities.formatDate(rowDate, Session.getScriptTimeZone(), "yyyy-MM-dd");
    } else {
      rowDateStr = String(rowDate);
    }
    
    if (data[i][2] === userId && rowDateStr === inputDateStr) {
      rowFound = i + 1;
      
      const ringkasan = data[i][17];
      if (ringkasan && String(ringkasan).length > 5) {
         currentKultumPoin = 10;
      }
      break;
    }
  }

  const finalPoin = checklistPoin + currentKultumPoin;
  
  // Mapping Kolom Legacy
  const colMap = { 
    'puasa': 6,    'sholat': 7,   'tarawih': 8,  
    'tadarus': 9,  'sedekah': 10, 'dzikir': 11   
  };

  if (rowFound > 0) {
    // --- UPDATE DATA LAMA ---
    for (let key in colMap) {
      sheet.getRange(rowFound, colMap[key]).setValue(checklistData[key] === true ? 'Ya' : 'Tidak');
    }
    sheet.getRange(rowFound, 12).setValue(catatan || ''); 
    sheet.getRange(rowFound, 13).setValue(finalPoin);
    sheet.getRange(rowFound, 19).setValue(checklistDataJSON); 
    
  } else {
    // --- INSERT DATA BARU ---
    const ibadahId = 'IBD' + new Date().getTime();
    const newRow = [
      ibadahId,                       
      tanggal,                        
      userId,                         
      schoolId,                       
      validation.data.classId,        
      checklistData['puasa'] ? 'Ya' : 'Tidak',   
      checklistData['sholat'] ? 'Ya' : 'Tidak',  
      checklistData['tarawih'] ? 'Ya' : 'Tidak', 
      checklistData['tadarus'] ? 'Ya' : 'Tidak', 
      checklistData['sedekah'] ? 'Ya' : 'Tidak', 
      checklistData['dzikir'] ? 'Ya' : 'Tidak',  
      catatan || '',                  
      finalPoin,                      
      getCurrentTime(),               
      '-', '-', '-', '-',             
      checklistDataJSON               
    ];
    sheet.appendRow(newRow);
  }

  // 4. TRIGGER GAMIFIKASI
  SpreadsheetApp.flush();
  let newBadges = [];
  try {
    newBadges = checkAndAwardBadges(userId, schoolId);
  } catch (e) {
    console.error("Gamification Error: " + e.toString());
  }

  return { 
    success: true, 
    message: 'Checklist berhasil disimpan.', 
    totalPoin: finalPoin,
    newBadges: newBadges 
  };
}

function getChecklistData(token, tanggal) {
  const validation = validateToken(token);
  if (!validation.valid) return { success: false, error: validation.error };
  
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // ============================================
    // 1. CEK TANGGAL MULAI RAMADHAN (LOGIKA BARU)
    // ============================================
    const settingsSheet = ss.getSheetByName('SETTINGS');
    if (settingsSheet) {
       const settingsData = settingsSheet.getDataRange().getValues();
       let ramadhanStart = null;
       
       for (let i = 1; i < settingsData.length; i++) {
         if (settingsData[i][0] === 'ramadhan_start_date') {
           ramadhanStart = new Date(settingsData[i][1]);
           ramadhanStart.setHours(0,0,0,0); // Reset jam
           break;
         }
       }
       
       // Jika admin sudah set tanggal, validasi tanggal input user
       if (ramadhanStart) {
          const requestDate = new Date(tanggal);
          requestDate.setHours(0,0,0,0);
          
          // JIKA TANGGAL DIMINTA < TANGGAL MULAI:
          // Kembalikan kosong seolah-olah tidak ada data
          if (requestDate < ramadhanStart) {
             return { success: true, found: false, data: {}, totalPoin: 0, catatan: '' };
          }
       }
    }
    // ============================================

    const sheet = ss.getSheetByName('IBADAH');
    const data = sheet.getDataRange().getValues();
    const userId = validation.data.userId;

    // Cari data user pada tanggal tersebut
    for (let i = 1; i < data.length; i++) {
      let rowDate = data[i][1];
      if (rowDate instanceof Date) {
        rowDate = Utilities.formatDate(rowDate, Session.getScriptTimeZone(), "yyyy-MM-dd");
      }
      
      if (data[i][2] === userId && rowDate === tanggal) {
        // Data Ditemukan!
        const dynamicDataRaw = data[i][18]; // Kolom S (Index 18)
        const totalPoin = data[i][12];      // Kolom M (Index 12)
        const catatan = data[i][11];        // Kolom L (Index 11)
        
        let checklistData = {};
        try {
          if (dynamicDataRaw && dynamicDataRaw !== '-') {
            checklistData = JSON.parse(dynamicDataRaw);
          }
        } catch (e) {
          checklistData = {};
        }
        
        return { 
          success: true, 
          found: true, 
          data: checklistData, 
          totalPoin: totalPoin,
          catatan: catatan
        };
      }
    }
    
    // Jika tidak ada data hari ini
    return { success: true, found: false, data: {}, totalPoin: 0, catatan: '' };

  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

// HELPER INTERNAL (WAJIB ADA UNTUK DATA HISTORY)
// ============================================
function getIbadahHistoryInternal(targetUserId, ss) {
    const sheetIbadah = ss.getSheetByName('IBADAH');
    const sheetKultum = ss.getSheetByName('KULTUM');
    const dataIbadah = sheetIbadah.getDataRange().getValues();
    const dataKultum = sheetKultum ? sheetKultum.getDataRange().getValues() : [];
    
    const historyMap = {};

    // 1. Proses Checklist Ibadah
    for (let i = 1; i < dataIbadah.length; i++) {
      if (dataIbadah[i][2] === targetUserId) {
        let tgl = dataIbadah[i][1];
        if (tgl instanceof Date) tgl = Utilities.formatDate(tgl, Session.getScriptTimeZone(), "yyyy-MM-dd");
        
        let dynamic = {};
        try { dynamic = JSON.parse(dataIbadah[i][18]); } catch(e) {}

        historyMap[tgl] = {
          tanggal: tgl,
          totalPoin: Number(dataIbadah[i][12]),
          checklist: dynamic,
          catatan: dataIbadah[i][11],
          penceramah: '-', judul: '-', hasKultum: false
        };
      }
    }
    
    // 2. Proses Jurnal Kultum
    for (let i = 1; i < dataKultum.length; i++) {
      if (dataKultum[i][2] === targetUserId) {
        let tgl = dataKultum[i][1];
        if (tgl instanceof Date) tgl = Utilities.formatDate(tgl, Session.getScriptTimeZone(), "yyyy-MM-dd");
        
        const poinKultum = Number(dataKultum[i][9]);
        
        if (historyMap[tgl]) {
          // Merge ke tanggal yang sudah ada
          historyMap[tgl].totalPoin += poinKultum;
          historyMap[tgl].penceramah = dataKultum[i][5];
          historyMap[tgl].judul = dataKultum[i][6];
          historyMap[tgl].hasKultum = true;
        } else {
          // Buat entri baru jika hanya ada kultum
          historyMap[tgl] = {
            tanggal: tgl, totalPoin: poinKultum, checklist: {}, catatan: '-',
            penceramah: dataKultum[i][5], judul: dataKultum[i][6], hasKultum: true
          };
        }
      }
    }
    
    // Urutkan berdasarkan tanggal (Terlama ke Terbaru agar enak dibaca urutannya)
    return Object.values(historyMap).sort((a, b) => new Date(a.tanggal) - new Date(b.tanggal));
}

function markAnnouncementsAsRead(token, announceIds) {

          const readId = 'READ' + new Date().getTime() + Math.floor(Math.random()*1000);
          sheet.appendRow([readId, id, userId, now]);
       }
    });

    return { success: true };

    return { success: false, error: e.toString() };
  }

function getAnnouncementReaders(token, announceId) {
  const validation = validateToken(token);
  if (!validation.valid || !['admin','walikelas','superadmin'].includes(validation.data.role)) {
     return { success: false, error: 'Unauthorized' };
  }

  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const readSheet = ss.getSheetByName('ANNOUNCEMENT_READS');
    const userSheet = ss.getSheetByName('USERS');
    
    if(!readSheet) return { success: true, data: [] };

    const readData = readSheet.getDataRange().getValues();
    const userData = userSheet.getDataRange().getValues();

    // 1. Ambil UserID yang sudah baca announcement ini
    const readerIds = [];
    const readTimestamps = {}; // Map ID -> Waktu Baca

    for(let i=1; i<readData.length; i++) {
       if(readData[i][1] === announceId) {
          const uid = readData[i][2];
          readerIds.push(uid);
          readTimestamps[uid] = formatDateForDisplay(readData[i][3]);
       }
    }

    // 2. Ambil Detail User (Nama & Kelas)
    const readers = [];
    for(let i=1; i<userData.length; i++) {
       const uid = userData[i][0];
       if(readerIds.includes(uid)) {
          readers.push({
             name: userData[i][2],
             nis: userData[i][1],
             classId: userData[i][6],
             readTime: readTimestamps[uid]
          });
       }
    }

    return { success: true, data: readers };

  } catch (e) {
    return { success: false, error: e.toString() };
  }
}

            isRelevant = true;
         }
         else if (rowType === 'school' && rowTarget === mySchoolId) {
             // Tipe Sekolah: Target harus SchoolID saya (Broadcast 1 Sekolah)
             isRelevant = true;
         }
         else if (rowType === 'global') {
             // Tipe Global: Semua user bisa melihat (biasanya dari Superadmin)
             isRelevant = true;
         }

         // C. Masukkan ke List jika Relevan
         if (isRelevant) {
            list.push({
              id: annId,
              date: formatDateForDisplay(data[i][1]),
              title: data[i][2],
              content: data[i][3],
              sender: data[i][6], // Nama Pengirim
              role: data[i][7],   // Role Pengirim
              isRead: readMap.has(annId) // Status: true/false
            });
         }
      }
      
      // Limitasi: Hanya ambil 10 pengumuman terbaru
      if (list.length >= 10) break; 
    }

    return { success: true, data: list };
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}
// 4. GET ANNOUNCEMENTS HISTORY (Untuk Admin/Wali Kelas melihat apa yang mereka kirim)
function getSentAnnouncements(token) {
  const validation = validateToken(token);
  if (!validation.valid) return { success: false, error: validation.error };

  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName('ANNOUNCEMENTS');
    if (!sheet) return { success: true, data: [] };

    const data = sheet.getDataRange().getValues();
    const list = [];

    for (let i = data.length - 1; i >= 1; i--) {
      // Admin Sekolah melihat semua di sekolahnya, Wali Kelas melihat semua di sekolahnya (atau bisa difilter by sender jika mau strict)
      // Disini kita beri akses lihat semua pengumuman di sekolah tsb agar transparan
      if (data[i][8] === validation.data.schoolId) {
         list.push({
           id: data[i][0],
           date: formatDateForDisplay(data[i][1]),
           title: data[i][2],
           content: data[i][3],
           type: data[i][4],
           target: data[i][5],
           sender: data[i][6]
         });
      }
      if (list.length >= 20) break; // Limit 20 history
    }
    return { success: true, data: list };

  } catch (e) {
    return { success: false, error: e.toString() };
  }
}

// 5. DELETE ANNOUNCEMENT
function deleteAnnouncement(token, annId) {
   const validation = validateToken(token);
   if (!validation.valid || !['admin','walikelas','superadmin'].includes(validation.data.role)) {
      return { success: false, error: 'Unauthorized' };
   }

   try {
     const ss = SpreadsheetApp.getActiveSpreadsheet();
     const sheet = ss.getSheetByName('ANNOUNCEMENTS');
     const data = sheet.getDataRange().getValues();
     
     for (let i = 0; i < data.length; i++) {
       if (data[i][0] === annId) {
         sheet.deleteRow(i + 1);
         return { success: true, message: 'Pengumuman dihapus' };
       }
     }
     return { success: false, error: 'ID tidak ditemukan' };
   } catch(e) {
     return { success: false, error: e.toString() };
   }
}



// 2. Cek & Berikan Badge
function checkAndAwardBadges(userId, schoolId) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const ibadahSheet = ss.getSheetByName('IBADAH');
  const badgeSheet = ss.getSheetByName('USER_BADGES');
  
  const ibadahData = ibadahSheet.getDataRange().getValues(); // Header row 0
  const existingBadges = badgeSheet.getDataRange().getValues();
  
  // Ambil Badge yang sudah dimiliki user
  const userOwnedBadges = new Set();
  for(let i=1; i<existingBadges.length; i++) {
     if(existingBadges[i][1] === userId) {
        userOwnedBadges.add(existingBadges[i][2]);
     }
  }

  const newBadges = [];

  // --- LOGIKA BADGE 1: "FULL FASTING" (30 Hari Puasa) ---
  if (!userOwnedBadges.has('full_fasting')) {
     let puasaCount = 0;
     for(let i=1; i<ibadahData.length; i++) {
        if(ibadahData[i][2] === userId && ibadahData[i][5] === 'Ya') { // Kolom F = Puasa
           puasaCount++;
        }
     }
     if(puasaCount >= 30) {
        newBadges.push({ key: 'full_fasting', title: 'Full Fasting 30 Hari' });
     }
  }

  // --- LOGIKA BADGE 2: "EARLY BIRD" (7 Hari Subuh Berturut-turut) ---
  if (!userOwnedBadges.has('early_bird')) {
     // Ambil data subuh user, urutkan tanggal
     const subuhDates = ibadahData
        .filter(row => row[2] === userId && row[6] === 'Ya') // Kolom G = Sholat (Asumsi Subuh check validasi di JSON detail jika perlu, disini pakai global Sholat Ya)
        // Jika ingin spesifik Subuh, harus parse kolom DynamicData (col 18)
        .map(row => ({ date: new Date(row[1]).setHours(0,0,0,0), raw: row }))
        .sort((a, b) => b.date - a.date);
     
     // Cek Detail JSON untuk memastikan "Subuh: True"
     let consecutiveSubuh = 0;
     if(subuhDates.length >= 7) {
        let currentCheck = subuhDates[0].date;
        // Cek JSON entry terakhir
        let jsonValid = false;
        try { if(JSON.parse(subuhDates[0].raw[18]).subuh === true) jsonValid = true; } catch(e){}

        if(jsonValid) {
           consecutiveSubuh = 1;
           for(let i=1; i<subuhDates.length; i++) {
              const prevDate = subuhDates[i].date;
              const expected = new Date(currentCheck);
              expected.setDate(expected.getDate() - 1);
              
              if(prevDate === expected.getTime()) {
                 // Cek JSON lagi
                 let prevJsonValid = false;
                 try { if(JSON.parse(subuhDates[i].raw[18]).subuh === true) prevJsonValid = true; } catch(e){}
                 
                 if(prevJsonValid) {
                    consecutiveSubuh++;
                    currentCheck = prevDate;
                 } else break;
              } else break;
           }
        }
     }

     if(consecutiveSubuh >= 7) {
        newBadges.push({ key: 'early_bird', title: 'Early Bird (7 Hari Subuh)' });
     }
  }

  // --- SIMPAN BADGE BARU ---
  if (newBadges.length > 0) {
     const dateNow = new Date();
     newBadges.forEach(b => {
        const id = 'BDG' + new Date().getTime() + Math.floor(Math.random()*100);
        badgeSheet.appendRow([id, userId, b.key, dateNow]);
        
        // Log Activity
        logActivity(userId, 'siswa', 'Badge Earned', schoolId, `Earned: ${b.title}`);
     });
     return newBadges; // Return untuk notifikasi frontend
  }
  
  return [];
}

function getCurrentTime() {
